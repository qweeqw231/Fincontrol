### 3.3 加 `friendlyError()` 工具（修复 GLOBAL-007）

**改文件**：`fincontrol-frontend/src/utils/formatters.js`

**末尾追加**：

```js
/**
 * 把 axios error 转换为面向用户的友好文案（PR 1 · 修复 GLOBAL-007）
 * - 避免暴露 axios stack / 业务码给最终用户
 * - 未知错误返回通用文案
 */
export function friendlyError(err) {
  if (err == null) return '未知错误'
  if (typeof err === 'string') return err
  // 业务错误：{ code, message }
  if (typeof err.code === 'number') {
    if (err.code === 0) return '网络异常，请检查后端服务'
    if (err.code >= 400 && err.code < 500) return '请求参数错误'
    if (err.code >= 500) return '服务器错误，请稍后重试'
    if (err.message) return err.message
    return '请求失败'
  }
  // axios 错误：err.message 通常含 stack 信息，回退到通用文案
  if (err.message && err.message.length < 80 && !err.message.includes('\n')) {
    return err.message
  }
  return '加载失败，请稍后重试'
}
```

**导出**：从 formatters.js 导出 `friendlyError`

**HomePage 引入并使用**（见 §3.2）

**新文件 4（拆分后的实用工具）**：`fincontrol-frontend/src/utils/blob.js`

```js
/**
 * 释放单个 blob URL（PR 1 · 修复 GLOBAL-015）
 * @param {{url?: string} | null | undefined} preview
 */
export function revokeOne(preview) {
  if (!preview || !preview.url) return
  try {
    URL.revokeObjectURL(preview.url)
  } catch (_) {
    // ignore
  }
}

/**
 * 批量释放 blob URL（PR 1 · 修复 GLOBAL-015）
 * @param {Array<{url?: string}> | null | undefined} previews
 */
export function revokeAll(previews) {
  if (!Array.isArray(previews)) return
  previews.forEach(revokeOne)
}
```

## 5. 风险与缓解

| 风险 | 缓解 |
| --- | --- |
| `useEffect(() => fetchLatest, [snap, fetchLatest])` 死循环 | `fetchLatest` 在 store 内已用 `useCallback` 或 zustand 不变引用；安全起见加 `if (!snap)` 保护 |
| StateShell 影响首页布局 | 独立 `.state-shell` 类（命名空间），不与 `.app-header` / `.hero-card` 冲突 |
| `friendlyError` 漏覆盖某类 axios 错误 | 单测覆盖 6 种场景（null / string / 0 / 4xx / 5xx / 长字符串） |
| `revokeAll` 在 `confirm` 中顺序错乱导致预览空白 | 顺序：**先 setFiles([]) 清空 state**（React 异步），**再 revokeAll**（副作用）—— 注：实际 React 18 batching 下两者都同步安全 |
| unmount 时 cleanup 引发的 stale state warning | cleanup 用空依赖 `[]`，只在 unmount 触发一次 |
| vitest setup 不支持 jsdom URL.createObjectURL | 测试用 mock 替换，避免依赖 jsdom 实现 |
| `@testing-library/react` 未装 | 检查 `package.json` 确认已装；如未装，本 PR 推迟 |
| `react-router-dom` MemoryRouter 测试需求 | 已装；StateShell 测试用 MemoryRouter 包裹 |
